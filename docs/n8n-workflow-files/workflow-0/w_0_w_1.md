# A) W0 — Intake (Amazon Custom)

### A1) Normalize Payload (replace existing Normalize node)
- Accept flexible shapes (Amazon Custom / backend / mock)
- Extract optional **characterSpecs**, **bookSpecs**, **orderDetails**
- Locate and sanitize **dedication** across likely fields

**Code (drop into Code node: “Normalize Payload”)**
```js
// Accepts many shapes (Amazon Custom / backend forwarder / your mock)
const raw = $input.first()?.json || {};
const orderId = raw.amazonOrderId || raw.orderId || raw.id || 'UNKNOWN-ORDER';

function safe(o){ return (o && typeof o === 'object') ? o : {}; }
function coerceStr(v){ return (v==null) ? undefined : String(v); }
function num(v){ const n = Number(v); return Number.isFinite(n) ? n : undefined; }

// characterSpecs (optional)
const cs = safe(raw.characterSpecs);
const characterSpecs = {
  childName: coerceStr(cs.childName),
  skinTone: coerceStr(cs.skinTone),
  hairColor: coerceStr(cs.hairColor),
  hairStyle: coerceStr(cs.hairStyle),
  age: num(cs.age),
  pronouns: coerceStr(cs.pronouns),
  favoriteColor: coerceStr(cs.favoriteColor),
  animalGuide: coerceStr(cs.animalGuide),
  clothingStyle: coerceStr(cs.clothingStyle),
};

// bookSpecs (optional)
const bs = safe(raw.bookSpecs);
const bookSpecs = {
  title: coerceStr(bs.title),
  totalPages: num(bs.totalPages),
  format: coerceStr(bs.format),
  bookType: coerceStr(bs.bookType), // 'animal-guide' | 'mvp-simple' | etc.
};

// orderDetails (optional)
const od = safe(raw.orderDetails);
const shipping = safe(od.shippingAddress);
const orderDetails = {
  quantity: num(od.quantity) || 1,
  shippingAddress: {
    name: coerceStr(shipping.name),
    address: coerceStr(shipping.address),
    city: coerceStr(shipping.city),
    state: coerceStr(shipping.state),
    zip: coerceStr(shipping.zip),
  }
};

// dedication (search across likely fields)
let dedication = '';
if (raw.dedication) dedication = String(raw.dedication);
if (!dedication && Array.isArray(raw.items)) {
  for (const it of raw.items) {
    const fields = it.customizations || it.customizationFields || it.options || [];
    for (const f of fields) {
      const n = String(f.name || f.label || f.type || '').toLowerCase();
      const v = String(f.value || f.text || f.stringValue || '');
      if ((/dedication|message|note/).test(n) && v.trim()) { dedication = v; break; }
    }
    if (dedication) break;
  }
}

// sanitize dedication
const MAX_CHARS = 500, MAX_LINES = 6;
let t = dedication.replace(/[\u0000-\u0008\u000B-\u001F\u007F]/g,'')
  .replace(/\r\n?/g,'\n').replace(/[ \t]{2,}/g,' ');
t = t.split('\n').map(l => l.trimEnd()).join('\n').trim();
if (t.length > MAX_CHARS) t = t.slice(0, MAX_CHARS).trim();
t = t.split('\n').slice(0, MAX_LINES).join('\n');
const esc = s => s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');

return [{
  json: {
    orderId,
    status: (raw.status ? String(raw.status) : 'queued_for_processing'),
    orderDate: raw.orderDate ? String(raw.orderDate) : new Date().toISOString(),
    customerEmail: coerceStr(raw.customerEmail),
    characterSpecs,
    bookSpecs,
    orderDetails,
    dedication: { raw: dedication, text: t, htmlSafe: esc(t) }
  }
}];
```
