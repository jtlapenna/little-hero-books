async function run(workflowId,payload,entry) {
  const capture=e=>{
    entry.executionId=String(e.id);entry.status=e.status;entry.nodes=Object.keys(e.data?.resultData?.runData||{});entry.httpCode=e.data?.resultData?.error?.httpCode||null;
    const description=[e.data?.resultData?.error?.description,...(e.data?.resultData?.error?.messages||[])].join('\n');
    entry.ownershipRejection=/claim is stale|does not own|current submission/.test(description);save();return e;
  };
  if(entry.webhookConsumed){assert.ok(resumeNegative&&entry.executionId,'Unknown outcome requires reconciliation');const old=await n8n(`executions/${entry.executionId}?includeData=true`);assert.equal(old.workflowId,workflowId);assert.equal(old.status,'error');entry.reconciledReadOnly=true;return capture(old);}
  const previous=await n8n(`executions?workflowId=${workflowId}&limit=100`);
  const existing=new Set(previous.data.map(x=>String(x.id)));
  entry.intentRecordedAt=new Date().toISOString();entry.webhookConsumed=true;save();
  const webhook=(await n8n(`workflows/${workflowId}`)).nodes.find(n=>n.type==='n8n-nodes-base.webhook').parameters.path;
  const ack=await jsonRequest(`https://thepeakbeyond.app.n8n.cloud/webhook/${webhook}`,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(payload)});
  assert.equal(ack.status,'accepted');
  const deadline=Date.now()+600_000;
  while(Date.now()<deadline) {
    const list=await n8n(`executions?workflowId=${workflowId}&limit=100`);
    const candidates=list.data.filter(x=>!existing.has(String(x.id)));
    assert.ok(candidates.length<=1,'Concurrent workflow execution requires reconciliation');
    if(candidates.length) {
      const e=await n8n(`executions/${candidates[0].id}?includeData=true`);
      entry.executionId=String(e.id);entry.status=e.status;save();
      if(['success','error','canceled','crashed'].includes(e.status)) {
        return capture(e);
      }
    }
    await new Promise(resolve=>setTimeout(resolve,5000));
  }
  throw new Error('Execution outcome unknown; do not resend');
}
async function verifyGraph(submission,items,entry) {
  const pending=[submission.accepted_root_order_id],visited=new Set(),graph=[];
  assert.ok(pending[0]);
  while(pending.length) {
    const id=pending.shift();if(visited.has(id))continue; assert.ok(visited.size<100);visited.add(id);
    const raw=await graphGet(id);
    assert.equal(raw.id,id);assert.equal(raw.orderType,'draft');assert.equal(raw.financialStatus,'draft');assert.equal(raw.fulfillmentStatus,'draft');assert.equal(raw.orderReferenceId,submission.order_reference_id);
    graph.push({id,orderType:raw.orderType,financialStatus:raw.financialStatus,fulfillmentStatus:raw.fulfillmentStatus,items:raw.items.map(item=>({itemReferenceId:item.itemReferenceId,quantity:item.quantity,pageCount:item.pageCount}))});
    pending.push(...(raw.connectedOrderIds||[]));
  }
  const actual=graph.flatMap(node=>node.items).map(x=>x.itemReferenceId).sort();
  assert.deepEqual(actual,items.map(x=>x.item_reference_id).sort());
  for(const item of graph.flatMap(node=>node.items)){assert.equal(item.quantity,1);assert.equal(item.pageCount,32);}
  const persisted=await rest(`print_provider_submission_orders?submission_id=eq.${submission.id}&select=provider_order_id,order_type,status`);
  assert.deepEqual(persisted.map(x=>x.provider_order_id).sort(),[...visited].sort()); assert.ok(persisted.every(x=>x.order_type==='draft'));
  entry.graph=graph;entry.persistedGraphVerified=true;save();
}
