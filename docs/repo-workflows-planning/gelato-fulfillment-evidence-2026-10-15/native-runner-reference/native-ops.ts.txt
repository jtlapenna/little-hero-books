export function executionGraph(execution:any,template:any){
 const actual=structuredClone(graph(execution)),expected=graph(template);
 assert.equal(actual.nodes.length,expected.nodes.length);
 const defaults:Record<string,Record<string,unknown>>={
  'n8n-nodes-base.webhook':{multipleMethods:false,authentication:'none',webhookNotice:''},
  'n8n-nodes-base.respondToWebhook':{enableResponseOutput:false,generalNotice:'',webhookNotice:''},
  'n8n-nodes-base.code':{mode:'runOnceForAllItems',language:'javaScript',notice:''},
  'n8n-nodes-base.httpRequest':{preBuiltAgentsCalloutHttpRequest:'',curlImport:'',provideSslCertificates:false,sendQuery:false,sendHeaders:false,contentType:'json',infoMessage:''},
 };
 for(let i=0;i<actual.nodes.length;i++){const n=actual.nodes[i],e=expected.nodes[i];assert.equal(n.id,e.id);assert.equal(n.type,e.type);
  for(const [key,value] of Object.entries(defaults[n.type]??{}))if(!(key in e.parameters)&&key in n.parameters){assert(isDeepStrictEqual(n.parameters[key],value),'Execution default differs: '+key);delete n.parameters[key];}
  if(n.type==='n8n-nodes-base.httpRequest'&&!('fullResponse' in (e.parameters.options?.response?.response??{}))&&'fullResponse' in (n.parameters.options?.response?.response??{})){assert.equal(n.parameters.options.response.response.fullResponse,false);delete n.parameters.options.response.response.fullResponse;}
 }
 assert(isDeepStrictEqual(actual,expected),'Execution graph differs beyond exact reviewed n8n defaults');return structuredClone(expected);
}

export function recoveryDisposition(s:any,now=Date.now()){
 if(s.request_state==='submitted'||s.request_state==='failed')return 'terminal';
 if(Date.parse(s.lease_expires_at)>now)return 'deferred-active-lease';
 if(s.action_reason==='reconciling')return 'unresolved';
 return 'eligible-or-held';
}
export function matchConsumedExecution(records:any[],event:any,known:any[]){
 const matched=[];for(const candidate of records){assert.equal(candidate.workflowId,event.data.copyId);assert(['success','error','canceled','crashed'].includes(candidate.status),'Execution still in flight; observe later without triggering');const body=candidate.data?.resultData?.runData?.['Webhook (Gelato Intake)']?.[0]?.data?.main?.[0]?.[0]?.json?.body;const digest=sha(body);assert(known.some(e=>e.data.inputSha256===digest),'Unowned execution on dedicated copy');if(digest===event.data.inputSha256)matched.push(candidate);}
 assert(matched.length<=1,'Multiple executions for exact consumed claim');return matched[0];
}
