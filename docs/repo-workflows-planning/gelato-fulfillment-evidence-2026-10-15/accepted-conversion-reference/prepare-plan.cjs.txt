'use strict';
const fs=require('node:fs'),crypto=require('node:crypto'),assert=require('node:assert/strict');const {Operator,digest,readPrivate,directory}=require('./operator.cjs');
const primary='<ORIGINATING_CHECKOUT>',native=primary+'/.gelato-testing/fulfillment-2026-10-15/native-20261016-C',previous=primary+'/.gelato-testing/fulfillment-2026-10-15/native-20261016-A',historical=primary+'/.gelato-testing/fulfillment-2026-10-15/native-20261016-B';
async function main(){
 assert(!fs.existsSync(directory+'/plan.json'),'Plan already exists, do not replace identities');
 const events=readPrivate(native+'/operations.jsonl').trim().split('\n').map(JSON.parse),fixture=JSON.parse(readPrivate(native+'/fixture-input-final.json')),pins=JSON.parse(readPrivate(native+'/approved-source.json'));
 const priorEvents=readPrivate(previous+'/operations.jsonl').trim().split('\n').map(JSON.parse),priorFixture=JSON.parse(readPrivate(previous+'/fixture-input-final.json')); const selected=[{c:fixture.cases.find(c=>c.kind==='website-single'),events},...priorFixture.cases.filter(c=>c.kind!=='website-single').map(c=>({c,events:priorEvents}))];
 const env=require(primary+'/back-end/node_modules/dotenv').parse(readPrivate(primary+'/.gelato-testing/.env.local'));const plan={at:new Date().toISOString(),keySha256:digest(env.GELATO_API_KEY),fixtureSha256:digest(selected.map(x=>x.c)),nativeJournalSha256:digest(readPrivate(native+'/operations.jsonl')),cases:[],orderKeys:selected.flatMap(x=>x.c.orders.map(o=>o.orderId)),workflowIds:['8ZqgZ7ksY0Fj8Acx','d7wi1rS0CxGVdRVb'],workflowGraphs:{}};
 assert.equal(plan.keySha256,JSON.parse(readPrivate(primary+'/.gelato-testing/fulfillment-2026-10-15/retained-evidence.json')).apiKeyFingerprint);
 plan.workflowIds.forEach((id,i)=>plan.workflowGraphs[id]=pins.normalGraphs[i]);for(const e of [...events,...priorEvents,...readPrivate(historical+'/operations.jsonl').trim().split('\n').map(JSON.parse)].filter(e=>e.kind==='verified'&&e.key.startsWith('copy:'))){plan.workflowIds.push(e.data.id);plan.workflowGraphs[e.data.id]=e.data.graphSha256;}
 assert.equal(plan.workflowIds.length,8);const op=new Operator(env,plan,{},null,{assert:()=>{throw Error('Preparation is read-only');}});
 for(const {c,events:caseEvents} of selected){const proofs=caseEvents.map(e=>e.kind==='native-proof'?e.data:e.kind==='observed'?e.data.proof:null).filter(p=>p?.kind===c.kind);assert(proofs.length);const proof=proofs.at(-1);assert.equal(proof.providerRoots.length,1);
  const members=await op.db('print_provider_submission_items','select=*&submission_id=eq.'+proof.submissionId,'child_order_id');assert.equal(members.length,c.orders.length);
  const artifacts=members.map(m=>{const a=proof.artifacts.find(a=>a.childId===m.child_order_id);assert(a&&a.providerStoredBytesMatch);assert.equal(m.provider_order_id,proof.providerRoots[0]);assert(proof.rawProviderOrders[0].items.some(i=>i.id===m.provider_item_id&&i.itemReferenceId===m.item_reference_id));assert(m.artifact_key.endsWith('/'+m.artifact_fingerprint+'.pdf'));return{child_order_id:m.child_order_id,artifact_key:m.artifact_key,provider_item_id:m.provider_item_id,sha256:a.sha256};});
  plan.cases.push({kind:c.kind,authorizationId:crypto.randomUUID(),submissionId:proof.submissionId,rootOrderId:c.orders[0].rootOrderId,providerRootId:proof.providerRoots[0],artifacts,workflowExecutionId:proof.executionId,shippingTier:proof.shippingTier});
 }
 fs.writeFileSync(directory+'/plan.json',JSON.stringify(plan,null,2)+'\n',{flag:'wx',mode:0o600});const closure={authorizationIds:plan.cases.map(c=>c.authorizationId),native:JSON.parse(readPrivate(native+'/closure-targets.json')),previousNative:[JSON.parse(readPrivate(previous+'/closure-targets.json')),JSON.parse(readPrivate(historical+'/closure-targets.json'))]};fs.writeFileSync(directory+'/closure-targets.json',JSON.stringify(closure,null,2)+'\n',{flag:'wx',mode:0o600});console.log(JSON.stringify({plannedCases:plan.cases.length,artifacts:plan.cases.flatMap(c=>c.artifacts).length,workflowPins:plan.workflowIds.length,liveWrites:0,providerCalls:0}));
}
main().catch(()=>{console.error('Plan preparation failed; preserve files; no authorization inserted');process.exitCode=1;});
