function readPrivate(p){const s=fs.lstatSync(p);assert(s.isFile()&&!s.isSymbolicLink()&&s.uid===process.getuid()&&(s.mode&0o077)===0,'Private file ownership/mode required');return fs.readFileSync(p,'utf8');}
async function save(p,data){const h=await fsp.open(p,'wx',0o600);try{await h.writeFile(JSON.stringify(data,null,2)+'\n');await h.sync();}finally{await h.close();}const d=await fsp.open(directory,'r');try{await d.sync();}finally{await d.close();}}
class Journal{
 constructor(){this.events=[];}
 async open(){const p=directory+'/operations.jsonl';try{const raw=readPrivate(p);assert(raw===''||raw.endsWith('\n'),'Torn journal');this.events=raw.trimEnd().split('\n').filter(Boolean).map(JSON.parse);}catch(e){if(e.code!=='ENOENT')throw e;}this.h=await fsp.open(p,'a',0o600);}
 find(kind,key){const a=this.events.filter(e=>e.kind===kind&&e.key===key);assert(a.length<=1,'Duplicate effect record');return a[0];}
 async record(kind,key,data){const e={at:new Date().toISOString(),kind,key,data};await this.h.writeFile(JSON.stringify(e)+'\n');await this.h.sync();this.events.push(e);}
 async effect(key,input,perform){assert(!this.find('intent',key),'Consumed intent: observe only, never repeat');await this.record('intent',key,{sha256:digest(input)});try{const result=await perform();await this.record('response',key,result);return result;}catch{await this.record('uncertain',key,{noRetry:true});throw Error('Effect uncertain; close and observe only');}}
 async close(){await this.h?.close();}
}
